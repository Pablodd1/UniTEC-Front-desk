<!-- UniTEC USA Design Chatbot Embed Code -->
<!-- Add this before the closing </body> tag on https://www.unitecusadesign.com/ -->

<script>
(function() {
  var container = document.createElement('div');
  container.style.cssText = 'position:fixed;bottom:20px;right:20px;width:480px;height:700px;z-index:9999;display:none;border-radius:20px;overflow:hidden;box-shadow:0 20px 60px rgba(0,0,0,0.3);border:1px solid rgba(255,255,255,0.1);';
  
  var iframe = document.createElement('iframe');
  iframe.src = 'https://unitec-front-desk.vercel.app/?embed=true&brandId=unitec&lang=es';
  iframe.style.cssText = 'width:100%;height:100%;border:none;border-radius:20px;';
  iframe.allow = 'microphone';
  iframe.title = 'UniTEC USA Design Chatbot';
  
  container.appendChild(iframe);
  document.body.appendChild(container);
  
  var button = document.createElement('button');
  button.innerHTML = '💬 Chat';
  button.style.cssText = 'position:fixed;bottom:20px;right:20px;z-index:10000;padding:16px 24px;background:linear-gradient(135deg,#0ea5e9 0%,#0284c7 100%);color:white;border:none;border-radius:50px;cursor:pointer;font-size:16px;font-weight:600;box-shadow:0 4px 20px rgba(14,165,233,0.4);display:flex;align-items:center;gap:8px;';
  document.body.appendChild(button);
  
  var isOpen = false;
  button.addEventListener('click', function() {
    isOpen = !isOpen;
    container.style.display = isOpen ? 'block' : 'none';
    button.style.display = isOpen ? 'none' : 'flex';
  });
  
  window.addEventListener('message', function(e) {
    if (e.data === 'close-chatbot') {
      isOpen = false;
      container.style.display = 'none';
      button.style.display = 'flex';
    }
  });
})();
</script>
